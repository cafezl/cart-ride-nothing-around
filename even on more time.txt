--- Cart Settings Tab (Added) ---
local CartSettings = Window:NewTab("Cart Settings")
local CartSettingsSection = CartSettings:NewSection("Cart Control")
 
local panicOn = false
 
-- Helper function to get the cart model
local function getCart()
    local char = game.Players.LocalPlayer.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or not hum.SeatPart then return nil end
    return hum.SeatPart.Parent
end
 
-- Panic Stop function
CartSettingsSection:NewToggle("Panic Stop", "Toggles anchoring the cart to stop it.", function(state)
    local cart = getCart()
    if not cart then return end
 
    panicOn = state
 
    for _, part in ipairs(cart:GetDescendants()) do
        if part:IsA("BasePart") then
            part.Anchored = state
            if state then
                part.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                part.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            end
        end
    end
end)
 
-- Teleport Cart to Checkpoint 1
CartSettingsSection:NewButton("Teleport to Checkpoint 1", "Teleports the cart to checkpoint 1.", function()
    local cart = getCart()
    if not cart or not cart.PrimaryPart then return end
 
    -- Coordinates from the Auto Win script for Checkpoint 1, with a small height adjustment
    local checkpoint1CFrame = CFrame.new(-430.898926, 164.75, 101.645676)
 
    -- Apply a 90-degree rotation to the CFrame
    local rotatedCFrame = checkpoint1CFrame * CFrame.Angles(0, math.rad(90), 0)
 
    cart:SetPrimaryPartCFrame(rotatedCFrame)
    cart.PrimaryPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    cart.PrimaryPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
end)
